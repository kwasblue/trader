const fmtUSD = (x: number | null | undefined) =>
  x === null || x === undefined
    ? "–"
    : x.toLocaleString(undefined, {
        style: "currency",
        currency: "USD",
        maximumFractionDigits: 2,
      });
const fmtPct = (x: number | null | undefined) =>
  x === null || x === undefined ? "–" : `${(x * 100).toFixed(2)}%`;
const fmtNum = (x: number | null | undefined, digits = 2) =>
  x === null || x === undefined
    ? "–"
    : x.toLocaleString(undefined, { maximumFractionDigits: digits });

// Mock PnL curve & equity for preview
function makeSeries(days = 240, start = 100000) {
  let cur = start;
  const out: Array<{ date: string; equity: number; pnl: number; ret: number }> = [];
  for (let i = days; i >= 0; i--) {
    const d = new Date();
    d.setDate(d.getDate() - i);
    const shock = (Math.random() - 0.5) * 0.01;
    cur *= 1 + shock;
    out.push({
      date: d.toISOString().slice(0, 10),
      equity: cur,
      pnl: cur - start,
      ret: cur / start - 1,
    });
  }
  return out;
}

// Mock realized PnL time series (daily realized and cumulative realized)
function makeRealizedSeries(days = 240) {
  let cum = 0;
  const out: Array<{ date: string; realized: number; cumRealized: number }> = [];
  for (let i = days; i >= 0; i--) {
    const d = new Date();
    d.setDate(d.getDate() - i);
    const daily = (Math.random() - 0.48) * 150; // daily realized
    cum += daily;
    out.push({ date: d.toISOString().slice(0, 10), realized: daily, cumRealized: cum });
  }
  return out;
}

// Build mock per-symbol daily PnL series
function symbolDailyPnl(symbol: string) {
  const n = 60;
  const out: Array<{ date: string; pnl: number }> = [];
  let cur = (Math.random() - 0.5) * 300;
  for (let i = n; i >= 0; i--) {
    const d = new Date();
    d.setDate(d.getDate() - i);
    cur += (Math.random() - 0.5) * 50;
    out.push({ date: d.toISOString().slice(0, 10), pnl: cur });
  }
  return out;
}

  const totalRealized = useMemo(() => symbols.reduce((a, s) => a + (s.realized || 0), 0), [symbols]);
  const longExposure = useMemo(() => symbols.filter((s) => s.side === "long").reduce((a, s) => a + s.qty * (s.last || 0), 0), [symbols]);

    if (!running) return;

      setEquityCurve((prev) => {
        const last = prev[prev.length - 1];
        const d = new Date();
        const shock = (Math.random() - 0.5) * 0.004;
        const nextEq = (last?.equity ?? 100000) * (1 + shock);
        return [
          ...prev.slice(1),
          {
            date: d.toISOString().slice(0, 10),
            equity: nextEq,
            pnl: nextEq - (prev[0]?.equity ?? 100000),

// RISK PANEL (higher contrast + semantic coloring)
// =====================================================
function RiskPanel({ totalUnreal, totalRealized, longExposure, shortExposure }: any) {
  const metrics = [
    { name: "Unrealized PnL", value: fmtUSD(totalUnreal), icon: SquareDot },
    { name: "Realized PnL", value: fmtUSD(totalRealized), icon: SquareDot },
    { name: "Long Exposure", value: fmtUSD(longExposure), icon: TrendingUp },
    { name: "Short Exposure", value: fmtUSD(shortExposure), icon: TrendingDown },
    { name: "Drawdown (day)", value: fmtPct(Math.random() * 0.05 - 0.03) },
    { name: "Risk Regime", value: ["low_vol", "normal_vol", "high_vol"][Math.floor(Math.random() * 3)] },
  ];

  function valueClass(name: string, value: string) {
    const neg = typeof value === "string" && value.trim().startsWith("-");
    const isPnL = /pnl/i.test(name) || /drawdown/i.test(name);
    const isExposure = /exposure/i.test(name);
    if (isPnL) return neg ? "text-red-400" : "text-emerald-400";
    if (isExposure) return neg ? "text-red-300" : "text-emerald-300";
    return "text-neutral-100";
  }

  return (
    <Card className="bg-neutral-900 border border-neutral-700">
      <CardHeader className="pb-2">
        <CardTitle className="text-neutral-50">Risk Panel</CardTitle>
      </CardHeader>
      <CardContent className="space-y-4">
        <div className="grid grid-cols-2 gap-3">
          {metrics.map((m, i) => (
            <div key={i} className="rounded-2xl border border-neutral-600 bg-neutral-950 p-3">
              <div className="text-xs text-neutral-200">{m.name}</div>
              <div className="mt-1 flex items-center gap-2 text-base font-bold">
                {m.icon ? <m.icon className="h-4 w-4 text-neutral-400" /> : null}
                <span className={valueClass(m.name, m.value as string)}>{m.value}</span>
              </div>
            </div>
          ))}
        </div>
        <Separator className="bg-neutral-700" />
        <OrderTicket />
      </CardContent>
    </Card>
  );
}

function SymbolCard({ s }: any) {
  const totalPnl = (s.unreal || 0) + (s.realized || 0);
  const series = useMemo(() => symbolDailyPnl(s.symbol), [s.symbol]);

  // regime pill color map (tinted backgrounds for readability)
  const regimeClass =
    ({
      low_vol: "bg-teal-500/15 text-teal-300 border-teal-700",
      normal_vol: "bg-neutral-700/30 text-neutral-200 border-neutral-600",
      high_vol: "bg-amber-500/15 text-amber-300 border-amber-700",
    } as Record<string, string>)[s.regime] || "bg-neutral-700/30 text-neutral-200 border-neutral-600";

  // risk score: ATR as % of last price, scaled 0–100
  const riskScore = (() => {
    const last = typeof s.last === "number" && s.last > 0 ? s.last : 1;
    const atr = typeof s.atr === "number" ? s.atr : 0;
    const pct = (atr / last) * 100;
    return Math.max(0, Math.min(100, Math.round(pct * 2)));
  })();

  const riskClass =
    riskScore > 60
      ? "bg-red-500/15 text-red-300 border-red-700"
      : riskScore > 40
      ? "bg-amber-500/15 text-amber-300 border-amber-700"
      : riskScore > 20
      ? "bg-teal-500/15 text-teal-300 border-teal-700"
      : "bg-emerald-500/15 text-emerald-300 border-emerald-700";

  return (
    <Card className="bg-neutral-900 border border-neutral-700">
      <CardHeader className="pb-2">
        <div className="flex items-center justify-between">
          <CardTitle className="text-neutral-50 flex items-center gap-2">
            {s.symbol}
            <Badge className={`border ${regimeClass}`}>{s.regime.replace("_", " ")}</Badge>
          </CardTitle>
          <div className="flex items-center gap-2">
            <Badge className={`border ${riskClass}`}>Risk {riskScore}</Badge>
            <Badge variant={s.side === "long" ? "default" : s.side === "short" ? "destructive" : "secondary"}>
              {s.side.toUpperCase()}
            </Badge>
          </div>
        </div>
      </CardHeader>
      <CardContent>
        <div className="grid grid-cols-2 gap-3">
          <KV label="Qty" value={fmtNum(s.qty, 0)} />
          <KV label="Avg" value={s.avg ? fmtUSD(s.avg) : "–"} />
          <KV label="Last" value={fmtUSD(s.last)} />
          <KV label="Unreal PnL" value={fmtUSD(s.unreal)} emphasis={s.unreal} />
          <KV label="Realized PnL" value={fmtUSD(s.realized)} emphasis={s.realized} />
          <KV label="Total PnL" value={fmtUSD(totalPnl)} emphasis={totalPnl} />
          <KV label="PnL %" value={fmtPct(s.pnlPct)} emphasis={s.pnlPct} />
          <KV label="ATR" value={fmtNum(s.atr)} />
          <KV label="Regime" value={s.regime} />
        </div>
        <div className="mt-3 h-24 w-full">
          <ResponsiveContainer width="100%" height="100%">
            <BarChart data={series} margin={{ left: 0, right: 0, top: 8, bottom: 0 }}>
              <XAxis dataKey="date" hide />
              <YAxis hide domain={["auto", "auto"]} />
              <RTooltip formatter={(v) => fmtUSD(v as number)} labelFormatter={(l) => `Date: ${l}`} />
              <Bar dataKey="pnl" />
            </BarChart>
          </ResponsiveContainer>
        </div>
      </CardContent>
    </Card>
  );
}

function KV({ label, value, emphasis }: { label: string; value: any; emphasis?: number }) {
  const isNeg = typeof emphasis === "number" && emphasis < 0;
  const isPos = typeof emphasis === "number" && emphasis > 0;
  return (
    <div className="rounded-xl border border-neutral-700 p-2 bg-neutral-950/60">
      <div className="text-[10px] uppercase tracking-wide text-neutral-300">{label}</div>
      <div className={`text-sm font-semibold ${isNeg ? "text-red-300" : isPos ? "text-emerald-300" : "text-neutral-100"}`}>{value}</div>
    </div>
  );
}

// BOT PERFORMANCE (historical performance tracking)
// =====================================================
function BotPerformance({ equityCurve, realizedCurve }: any) {
  // === Monthly aggregation ===
  const monthlyEquity = useMemo(() => {
    const byMonth: Record<string, { first: number; last: number }> = {};
    equityCurve.forEach((p: any) => {
      const m = p.date.slice(0, 7);
      if (!byMonth[m]) byMonth[m] = { first: p.equity, last: p.equity };
      byMonth[m].last = p.equity;
    });
    return Object.entries(byMonth).map(([month, { first, last }]) => ({ month, ret: last / first - 1 }));
  }, [equityCurve]);

  const monthlyRealized = useMemo(() => {
    const byMonth: Record<string, number> = {};
    realizedCurve.forEach((p: any) => {
      const m = p.date.slice(0, 7);
      byMonth[m] = (byMonth[m] || 0) + p.realized;
    });
    return Object.entries(byMonth).map(([month, realized]) => ({ month, realized }));
  }, [realizedCurve]);

  // === Stats ===
  const equityStats = useMemo(() => {
    const start = equityCurve[0]?.equity ?? 100000;
    const end = equityCurve[equityCurve.length - 1]?.equity ?? start;
    const totalRet = end / start - 1;
    let peak = -Infinity,
      maxDD = 0;
    equityCurve.forEach((p: any) => {
      peak = Math.max(peak, p.equity);
      maxDD = Math.min(maxDD, (p.equity - peak) / peak);
    });
    return { start: fmtUSD(start), end: fmtUSD(end), totalRet: fmtPct(totalRet), maxDD: fmtPct(maxDD) };
  }, [equityCurve]);

  const realizedStats = useMemo(() => {
    const total = realizedCurve.reduce((a: number, p: any) => a + p.realized, 0);
    const avgDaily = realizedCurve.length ? total / realizedCurve.length : 0;
    const wins = realizedCurve.filter((p: any) => p.realized > 0).length;
    const winRate = realizedCurve.length ? wins / realizedCurve.length : 0;
    return { total: fmtUSD(total), avgDaily: fmtUSD(avgDaily), winRate: fmtPct(winRate) };
  }, [realizedCurve]);

  return (
    <div className="grid grid-cols-1 gap-6 xl:grid-cols-3">
      {/* Equity Panel */}
      <Card className="bg-neutral-900 border border-neutral-700 xl:col-span-2">
        <CardHeader className="pb-2">
          <CardTitle className="text-neutral-50">Cumulative Equity</CardTitle>
        </CardHeader>
        <CardContent>
          <div className="h-72 w-full">
            <ResponsiveContainer width="100%" height="100%">
              <AreaChart data={equityCurve} margin={{ top: 10, right: 30, left: 0, bottom: 0 }}>
                <defs>
                  <linearGradient id="perfFill" x1="0" y1="0" x2="0" y2="1">
                    <stop offset="5%" stopOpacity={0.45} />
                    <stop offset="95%" stopOpacity={0} />
                  </linearGradient>
                </defs>
                <XAxis dataKey="date" tick={{ fill: "#e5e7eb" }} />
                <YAxis tick={{ fill: "#e5e7eb" }} tickFormatter={(v) => fmtNum(v as number, 0)} />
                <RTooltip formatter={(v) => fmtUSD(v as number)} labelFormatter={(l) => `Date: ${l}`} />
                <Area type="monotone" dataKey="equity" strokeWidth={2} fillOpacity={1} fill="url(#perfFill)" />
                <Line type="monotone" dataKey="equity" strokeWidth={2} dot={false} />
              </AreaChart>
            </ResponsiveContainer>
          </div>
        </CardContent>
      </Card>

      {/* Equity Stats */}
      <Card className="bg-neutral-900 border border-neutral-700">
        <CardHeader className="pb-2">
          <CardTitle className="text-neutral-50">Equity Stats</CardTitle>
        </CardHeader>
        <CardContent className="space-y-3">
          <div className="grid grid-cols-2 gap-3">
            <KV label="Starting Equity" value={equityStats.start} />
            <KV label="Ending Equity" value={equityStats.end} />
            <KV label="Total Return" value={equityStats.totalRet} emphasis={parseFloat(equityStats.totalRet) || 0} />
            <KV label="Max Drawdown" value={equityStats.maxDD} emphasis={parseFloat(equityStats.maxDD) || 0} />
          </div>
          <Separator className="bg-neutral-700" />
          <div className="text-sm font-semibold">Monthly Equity Returns</div>
          <div className="h-40 overflow-auto rounded-xl border border-neutral-700">
            <table className="w-full text-sm">
              <thead className="bg-neutral-900">
                <tr className="text-left">
                  <th className="px-3 py-2 font-semibold text-neutral-100">Month</th>
                  <th className="px-3 py-2 font-semibold text-neutral-100">Return</th>
                </tr>
              </thead>
              <tbody>
                {monthlyEquity.map((m: any) => (
                  <tr key={m.month} className="border-t border-neutral-700">
                    <td className="px-3 py-2 text-neutral-100">{m.month}</td>
                    <td className={`px-3 py-2 font-semibold ${m.ret < 0 ? "text-red-300" : "text-emerald-300"}`}>
                      {fmtPct(m.ret)}
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </CardContent>
      </Card>

      {/* Realized PnL Panel */}
      <Card className="bg-neutral-900 border border-neutral-700 xl:col-span-2">
        <CardHeader className="pb-2">
          <CardTitle className="text-neutral-50">Realized PnL (Daily & Cumulative)</CardTitle>
        </CardHeader>
        <CardContent>
          <div className="grid grid-cols-1 gap-4">
            <div className="h-48 w-full">
              <ResponsiveContainer width="100%" height="100%">
                <BarChart data={realizedCurve} margin={{ top: 10, right: 20, left: 0, bottom: 0 }}>
                  <XAxis dataKey="date" hide />
                  <YAxis tick={{ fill: "#e5e7eb" }} tickFormatter={(v) => fmtNum(v as number, 0)} />
                  <RTooltip formatter={(v) => fmtUSD(v as number)} labelFormatter={(l) => `Date: ${l}`} />
                  <Bar dataKey="realized" />
                </BarChart>
              </ResponsiveContainer>
            </div>
            <div className="h-48 w-full">
              <ResponsiveContainer width="100%" height="100%">
                <LineChart data={realizedCurve} margin={{ top: 0, right: 20, left: 0, bottom: 0 }}>
                  <XAxis dataKey="date" tick={{ fill: "#e5e7eb" }} />
                  <YAxis tick={{ fill: "#e5e7eb" }} tickFormatter={(v) => fmtNum(v as number, 0)} />
                  <RTooltip formatter={(v) => fmtUSD(v as number)} labelFormatter={(l) => `Date: ${l}`} />
                  <Line type="monotone" dataKey="cumRealized" strokeWidth={2} dot={false} />
                </LineChart>
              </ResponsiveContainer>
            </div>
          </div>
        </CardContent>
      </Card>

      {/* Realized Stats */}
      <Card className="bg-neutral-900 border border-neutral-700">
        <CardHeader className="pb-2">
          <CardTitle className="text-neutral-50">Realized PnL Stats</CardTitle>
        </CardHeader>
        <CardContent className="space-y-3">
          <div className="grid grid-cols-2 gap-3">
            <KV label="Total Realized" value={realizedStats.total} emphasis={parseFloat(realizedStats.total) || 0} />
            <KV label="Avg Daily" value={realizedStats.avgDaily} emphasis={parseFloat(realizedStats.avgDaily) || 0} />
            <KV label="Winning Days" value={realizedStats.winRate} emphasis={parseFloat(realizedStats.winRate) || 0} />
          </div>
          <Separator className="bg-neutral-700" />
          <div className="text-sm font-semibold">Monthly Realized PnL</div>
          <div className="h-40 overflow-auto rounded-xl border border-neutral-700">
            <table className="w-full text-sm">
              <thead className="bg-neutral-900">
                <tr className="text-left">
                  <th className="px-3 py-2 font-semibold text-neutral-100">Month</th>
                  <th className="px-3 py-2 font-semibold text-neutral-100">Realized</th>
                </tr>
              </thead>
              <tbody>
                {monthlyRealized.map((m: any) => (
                  <tr key={m.month} className="border-t border-neutral-700">
                    <td className="px-3 py-2 text-neutral-100">{m.month}</td>
                    <td className={`px-3 py-2 font-semibold ${m.realized < 0 ? "text-red-300" : "text-emerald-300"}`}>
                      {fmtUSD(m.realized as number)}
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </CardContent>
      </Card>
    </div>
  );
}
